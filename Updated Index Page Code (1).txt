<script type="module">
  import { initializeApp } from "https://www.gstatic.com/firebasejs/10.12.2/firebase-app.js";
  import { getAuth, onAuthStateChanged } from "https://www.gstatic.com/firebasejs/10.12.2/firebase-auth.js";
  import {
    getFirestore, doc, getDoc, collection, query, where, getDocs,
    deleteDoc, addDoc, runTransaction, Timestamp, serverTimestamp
  } from "https://www.gstatic.com/firebasejs/10.12.2/firebase-firestore.js";

  const ADMIN_UID = "DEBuc9USNcULaH9BY1tXvbRJZYn1";
  const firebaseConfig = {
    // Keep your existing Firebase config object here
  };

  const app = initializeApp(firebaseConfig);
  const auth = getAuth(app);
  const db = getFirestore(app);

  let currentUser = null;
  let currentUserData = null;
  let homeSettings = {
    rewardPerSuccess: 0.50,
    deductionPerFail: 0.25,
    depositRenewCost: 4.00
  };

  function hideLoadingScreen() {
    const loader = document.getElementById("appLoadingScreen");
    if (!loader) return;
    loader.style.opacity = "0";
    setTimeout(() => { loader.style.display = "none"; }, 250);
  }

  function switchView(viewName) {
    ["homeView", "notificationsView", "marketplaceView", "renewView"].forEach((id) => {
      document.getElementById(id)?.classList.add("hidden");
    });
    const targetMap = {
      home: "homeView",
      notifications: "notificationsView",
      marketplace: "marketplaceView",
      renew: "renewView"
    };
    document.getElementById(targetMap[viewName] || "homeView")?.classList.remove("hidden");
    document.getElementById("sideMenuDrawer")?.classList.add("hidden");
  }

  document.getElementById("menuOpenBtn").addEventListener("click", () => {
    document.getElementById("sideMenuDrawer").classList.remove("hidden");
  });
  document.getElementById("menuCloseBtn").addEventListener("click", () => {
    document.getElementById("sideMenuDrawer").classList.add("hidden");
  });
  document.getElementById("notifBellBtn").addEventListener("click", () => switchView("notifications"));
  document.querySelectorAll(".nav-link").forEach((btn) => {
    btn.addEventListener("click", () => switchView(btn.dataset.nav));
  });

  function renderBalanceAndDeposit() {
    const successful = Number(currentUserData?.successfulAnswers || 0);
    const unsuccessful = Number(currentUserData?.unsuccessfulAnswers || 0);
    const reward = Number(homeSettings.rewardPerSuccess || 0);
    const deduction = Number(homeSettings.deductionPerFail || 0);
    const deposit = Number(currentUserData?.depositBalance ?? 4.00);
    const renewCost = Number(homeSettings.depositRenewCost ?? 4.00);

    const calculatedBalance = (successful * reward) - (unsuccessful * deduction);
    document.getElementById("userBalanceValue").textContent = calculatedBalance.toFixed(2);
    document.getElementById("userDepositValue").textContent = deposit.toFixed(2);
    document.getElementById("renewPageDeposit").textContent = deposit.toFixed(2);
    document.getElementById("renewCostDisplay").textContent = renewCost.toFixed(2);
    document.getElementById("balanceFormulaHint").textContent =
      `(${successful} × ${reward.toFixed(2)}) - (${unsuccessful} × ${deduction.toFixed(2)})`;
  }

  async function loadHomeSettings() {
    const snap = await getDoc(doc(db, "siteSettings", "home"));
    if (snap.exists()) {
      const d = snap.data();
      homeSettings = {
        rewardPerSuccess: Number(d.rewardPerSuccess ?? 0.50),
        deductionPerFail: Number(d.deductionPerFail ?? 0.25),
        depositRenewCost: Number(d.depositRenewCost ?? 4.00)
      };
      if (d.headline) {
        document.getElementById("scrollingHeadlineText").textContent = d.headline;
      }
    }
  }

  // 3. Load & auto-delete notifications older than 48 hours
  async function loadNotifications(uid) {
    const now = Timestamp.now();
    const snap = await getDocs(collection(db, "notifications"));
    const listEl = document.getElementById("notificationsList");
    const badgeEl = document.getElementById("notifBadge");
    const activeItems = [];

    for (const docSnap of snap.docs) {
      const data = docSnap.data();
      const exp = data.expiresAt;
      if (exp && exp.toMillis() <= now.toMillis()) {
        try { await deleteDoc(doc(db, "notifications", docSnap.id)); } catch (_) {}
        continue;
      }
      if (!data.targetUid || data.targetUid === "all" || data.targetUid === uid) {
        activeItems.push({ id: docSnap.id, ...data });
      }
    }

    if (activeItems.length > 0) {
      badgeEl.textContent = String(activeItems.length);
      badgeEl.classList.remove("hidden");
      listEl.innerHTML = activeItems.map((item) => `
        <div class="bg-slate-900 border border-slate-800 rounded-xl p-4">
          <div class="font-semibold text-sm text-amber-300">${item.title || "Notification"}</div>
          <p class="text-xs text-slate-300 mt-1">${item.message || ""}</p>
        </div>
      `).join("");
    } else {
      badgeEl.classList.add("hidden");
      listEl.innerHTML = `<p class="text-sm text-slate-400">No active notifications (auto-deleted after 48 hours).</p>`;
    }
  }

  async function loadRenewHistory(uid) {
    const q = query(collection(db, "renewRequests"), where("uid", "==", uid));
    const snap = await getDocs(q);
    const listEl = document.getElementById("renewHistoryList");
    if (snap.empty) {
      listEl.innerHTML = `<p class="text-slate-400">No renewal history yet.</p>`;
      return;
    }
    const items = snap.docs.map((d) => d.data());
    listEl.innerHTML = items.map((r) => `
      <div class="flex items-center justify-between bg-slate-950 border border-slate-800 rounded-lg px-3 py-2">
        <span>${r.method || "Renewal"} — $${Number(r.amount || 0).toFixed(2)}</span>
        <span class="uppercase text-[11px] px-2 py-0.5 rounded bg-emerald-950 text-emerald-400">${r.status || "approved"}</span>
      </div>
    `).join("");
  }

  // 6. Renew Account with Deposit Balance + add history for Admin & Home
  document.getElementById("renewWithDepositBtn").addEventListener("click", async () => {
    const statusEl = document.getElementById("renewStatusMsg");
    if (!currentUser) {
      statusEl.textContent = "Please sign in first.";
      return;
    }
    const renewCost = Number(homeSettings.depositRenewCost ?? 4.00);
    const userRef = doc(db, "users", currentUser.uid);

    try {
      await runTransaction(db, async (tx) => {
        const userSnap = await tx.get(userRef);
        if (!userSnap.exists()) throw new Error("User profile not found.");
        const uData = userSnap.data();
        const currentDeposit = Number(uData.depositBalance || 0);
        if (currentDeposit < renewCost) {
          throw new Error(`Insufficient deposit balance (${currentDeposit.toFixed(2)} < ${renewCost.toFixed(2)}).`);
        }
        const nextDeposit = Number((currentDeposit - renewCost).toFixed(2));
        const baseTime = uData.validUntil?.toMillis && uData.validUntil.toMillis() > Date.now()
          ? uData.validUntil.toMillis()
          : Date.now();
        const nextValidUntil = Timestamp.fromMillis(baseTime + 30 * 24 * 60 * 60 * 1000);

        tx.update(userRef, {
          depositBalance: nextDeposit,
          validUntil: nextValidUntil,
          updatedAt: serverTimestamp()
        });
      });

      await addDoc(collection(db, "renewRequests"), {
        uid: currentUser.uid,
        method: "Deposit Balance",
        amount: renewCost,
        status: "approved",
        createdAt: serverTimestamp()
      });

      const refreshed = await getDoc(userRef);
      currentUserData = refreshed.data();
      renderBalanceAndDeposit();
      await loadRenewHistory(currentUser.uid);
      statusEl.textContent = `Account renewed successfully using ${renewCost.toFixed(2)} from Deposit Balance!`;
    } catch (err) {
      statusEl.textContent = err.message || "Could not renew account.";
    }
  });

  onAuthStateChanged(auth, async (user) => {
    currentUser = user;
    const adminBtn = document.getElementById("adminPanelBtn");
    if (user && user.uid === ADMIN_UID) {
      adminBtn.classList.remove("hidden");
    } else {
      adminBtn.classList.add("hidden");
    }

    await loadHomeSettings();
    if (user) {
      const uSnap = await getDoc(doc(db, "users", user.uid));
      currentUserData = uSnap.exists() ? uSnap.data() : {};
      renderBalanceAndDeposit();
      await loadNotifications(user.uid);
      await loadRenewHistory(user.uid);
    } else {
      renderBalanceAndDeposit();
    }
    hideLoadingScreen();
  });
</script>
