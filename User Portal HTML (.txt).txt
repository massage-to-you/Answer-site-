    function computeUserBalance(uData) {
      if (!uData) return 0;
      if (uData.balanceOverride !== undefined && uData.balanceOverride !== null && uData.balanceOverride !== "") {
        const ov = Number(uData.balanceOverride);
        if (Number.isFinite(ov)) return Math.max(0, ov);
      }
      const submitted = Math.max(0, Number(uData.submittedCount || 0));
      const fail = Math.min(submitted, Math.max(0, Number(uData.unsuccessfulAnswers ?? (Array.isArray(uData.unsuccessfulJobIds) ? uData.unsuccessfulJobIds.length : 0))));
      const successful = Math.max(0, submitted - fail);
      const reward = Number(siteSettingsCache.rewardPerSuccess || 0);
      const deduct = Number(siteSettingsCache.deductionPerFail || 0);
      return Math.max(0, (successful * reward) - (fail * deduct));
    }

    window.submitWithdrawRequest = async function(e) {
      e.preventDefault();
      if (isSubmittingWallet || !currentUser) return;
      const method = document.getElementById("wdMethod").value;
      const accountNumber = document.getElementById("wdAccountNumber").value.trim();
      const amount = Number(document.getElementById("wdAmount").value);
      const minW = Number(walletGatewayCache.minWithdraw || 0);
      const avail = computeUserBalance(currentUserData);

      if (!accountNumber || !Number.isFinite(amount) || amount <= 0) {
        showToast("Enter a valid account number and amount.");
        return;
      }
      if (minW > 0 && amount < minW) {
        showToast(`Minimum withdraw amount is ${fmtMoney(minW)} BDT.`);
        return;
      }
      if (amount > avail) {
        showToast("Insufficient balance for withdrawal.");
        return;
      }

      isSubmittingWallet = true;
      const btn = document.getElementById("submitWithdrawBtn");
      btn.disabled = true;
      try {
        const nextBal = Number((avail - amount).toFixed(2));
        await addDoc(collection(db, "depositWithdrawRequests"), {
          type: "withdraw",
          uid: currentUser.uid,
          loginId: currentUserData.loginId || currentUser.uid,
          method,
          accountNumber,
          amount: Number(amount.toFixed(2)),
          status: "pending",
          createdAt: new Date().toISOString()
        });
        await updateDoc(doc(db, "users", currentUserDocId || currentUser.uid), {
          balance: nextBal,
          balanceOverride: nextBal,
          withdrawnTotal: Number((Number(currentUserData.withdrawnTotal || 0) + amount).toFixed(2)),
          updatedAt: new Date().toISOString()
        });
        currentUserData.balance = nextBal;
        currentUserData.balanceOverride = nextBal;
        currentUserData.withdrawnTotal = Number((Number(currentUserData.withdrawnTotal || 0) + amount).toFixed(2));
        renderUserWallets();
        e.target.reset();
        await loadMyDwHistory();
        showToast("Withdraw request submitted.");
      } catch (err) {
        showToast("Withdraw failed: " + (err.message || err));
      } finally {
        isSubmittingWallet = false;
        btn.disabled = false;
      }
    };

    function getNotificationExpiryMs(n) {
      if (n.expiresAt?.toMillis) {
        const v = n.expiresAt.toMillis();
        if (Number.isFinite(v)) return v;
      }
      if (n.clientExpiresAt) {
        const p = Date.parse(n.clientExpiresAt);
        if (Number.isFinite(p)) return p;
      }
      const createdMs = getNotificationCreatedMs(n);
      return createdMs ? createdMs + FORTY_EIGHT_HOURS_MS : Date.now() + FORTY_EIGHT_HOURS_MS;
    }

    function getNotificationCreatedMs(n) {
      if (n.createdAt?.toMillis) {
        const v = n.createdAt.toMillis();
        if (Number.isFinite(v)) return v;
      }
      if (n.clientCreatedAt) {
        const p = Date.parse(n.clientCreatedAt);
        if (Number.isFinite(p)) return p;
      }
      return Date.now();
    }

    function renderNotificationsUI() {
      const now = Date.now();
      const uid = currentUser?.uid || "";
      const visible = activeNotifications
        .filter(n => {
          const exp = getNotificationExpiryMs(n);
          if (exp <= now) return false;
          const target = String(n.targetUid || "all");
          return target === "all" || (uid && target === uid);
        })
        .sort((a, b) => getNotificationCreatedMs(b) - getNotificationCreatedMs(a));

      const count = visible.length;
      const hBadge = document.getElementById("headerNotifBadge");
      const mBadge = document.getElementById("menuNotifBadge");
      [hBadge, mBadge].forEach(badge => {
        if (!badge) return;
        badge.textContent = String(count);
        badge.classList.toggle("has-items", count > 0);
      });

      const listEl = document.getElementById("notificationsHistoryList");
      if (!listEl) return;
      if (!count) {
        listEl.innerHTML = '<div class="form-help">No notifications in the last 48 hours.</div>';
        return;
      }
      listEl.innerHTML = visible.map(n => {
        const expMs = getNotificationExpiryMs(n);
        const hoursLeft = Math.max(1, Math.ceil((expMs - now) / (60 * 60 * 1000)));
        return `
          <div class="notif-card-item">
            <div class="notif-card-top">
              <div class="notif-card-title"><i class="fa-solid fa-bell"></i> ${escapeHtml(n.title || "Announcement")}</div>

            <div class="form-help" style="margin-top:8px;">Sent: ${escapeHtml(formatBdDate(n.createdAt || n.clientCreatedAt))}</div>
          </div>
        `;
      }).join("");
    }

      }, err => {

    function computeUserStats(uData, jobsList = []) {
      const uid = currentUser?.uid || "";
      const loginId = (uData?.loginId || "").toLowerCase();
      const submittedFromJobs = jobsList.filter(j => {
        const jUid = String(j.authorUID || "");
        const jLogin = String(j.authorLoginId || j.loginId || "").toLowerCase();
        return (uid && jUid === uid) || (loginId && (jLogin === loginId || jUid.toLowerCase() === loginId));
      }).length;
      const submitted = Math.max(Number(uData?.submittedCount || 0), submittedFromJobs);
      const failedIds = Array.isArray(uData?.unsuccessfulJobIds) ? uData.unsuccessfulJobIds.map(String) : [];
      const unsuccessful = Math.min(submitted, Math.max(Number(uData?.unsuccessfulAnswers || 0), failedIds.length));
      const successful = Math.max(0, submitted - unsuccessful);
      const rate = submitted > 0 ? Math.round((successful / submitted) * 100) : 0;
      return { submitted, unsuccessful, successful, rate };
    }

    function checkAccountExpiration(userData) {
      if (!userData || !userData.validUntil) return false;
      const raw = String(userData.validUntil).trim();
      if (!raw || raw.toLowerCase() === "not set") return false;
      const parsed = new Date(raw);
      if (Number.isNaN(parsed.getTime())) return false;
      const expiryEnd = new Date(parsed.getFullYear(), parsed.getMonth(), parsed.getDate(), 23, 59, 59, 999);
      return Date.now() > expiryEnd.getTime();

        hideSmoothReloadSpinner();

        return;

      currentUserData = userData || {};

      document.getElementById("ui-user-uid").textContent = "ID: " + loginIdDisplay;

    window.triggerSearch = async function() {
      if (isCurrentUserExpired) {
        switchView("renew", document.getElementById("nav-renew"));
        return;
      }
      const rawQuery = (document.getElementById("searchInput").value || "").trim();
      const resultsSection = document.getElementById("sec-results");
      const container = document.getElementById("resultContainer");

      if (!rawQuery) {
        resultsSection.classList.remove("search-active");
        container.innerHTML = "";
        return;
      }

      await loadAllJobs();
      const qNorm = normalizeJobId(rawQuery);
      const qLower = rawQuery.toLowerCase();

      const matches = allJobsCache.filter(j => {
        const jId = normalizeJobId(j.jobId || j.id);
        const jName = String(j.jobName || "").toLowerCase();
        return jId === qNorm || jId.includes(qNorm) || jName.includes(qLower);
      });

      resultsSection.classList.add("search-active");
      if (!matches.length) {
        container.innerHTML = `<div class="form-help" style="text-align:center;padding:18px;">No matching answer found for "${escapeHtml(rawQuery)}".</div>`;
        return;
      }

      container.innerHTML = matches.map(j => {
        const jobId = j.jobId || j.id;
        const jobName = j.jobName || "Untitled Job";
        const ansText = formatAnswerDisplay(j.answer || "");
        const imgSrc = j.image || j.imageUrl || "";
        const ytLink = j.youtubeLink || "";
        const dislikes = Number(j.dislikes || 0);
        const sender = j.authorLoginId || j.authorUID || "User";

        return `
          <div class="result-card">
            ${imgSrc ? `<div class="result-img-box"><img src="${escapeHtml(imgSrc)}" alt="${escapeHtml(jobName)}" loading="lazy"></div>` : `<div class="result-img-box" style="display:flex;align-items:center;justify-content:center;color:var(--text-muted);font-size:12px;">No Image</div>`}
            <div class="result-details">
              <span class="job-id-tag">Job ID: ${escapeHtml(jobId)}</span>
              <h3>${escapeHtml(jobName)}</h3>
              <div class="result-sender"><i class="fa-solid fa-user-check"></i> Submitted by: ${escapeHtml(sender)}</div>
              <div class="answer-container">
                <span class="correct-answer-label">Correct Answer:</span>
                <div class="answer-text">${escapeHtml(ansText)}</div>
              </div>
              <div class="card-actions">
                <button type="button" class="copy-btn" onclick="copyAnswerText(${JSON.stringify(ansText)})"><i class="fa-regular fa-copy"></i> Copy Answer</button>
                ${ytLink ? `<a href="${escapeHtml(ytLink)}" target="_blank" rel="noopener noreferrer" class="yt-btn"><i class="fa-brands fa-youtube"></i> Watch Video</a>` : ""}
                <button type="button" class="dislike-btn" onclick="openDislikeModal('${escapeHtml(j.id)}', '${escapeHtml(jobId)}', '${escapeHtml(jobName)}')">
                  <i class="fa-solid fa-thumbs-down"></i> Dislike <span class="dislike-count-badge">${dislikes}/3</span>
                </button>
              </div>
            </div>
          </div>
        `;
      }).join("");
    };

    window.submitDislikeSuggestion = async function(e) {
      e.preventDefault();
      if (isSubmittingDislike || !currentUser) return;
      const docId = document.getElementById("dislike-job-id").value.trim();
      const suggestion = document.getElementById("dislike-suggestion").value.trim();
      if (!docId || !suggestion) return;

      isSubmittingDislike = true;
      const btn = document.getElementById("submitDislikeBtn");
      btn.disabled = true;

      try {
        const jobRef = doc(db, "jobs", docId);
        const snap = await getDoc(jobRef);
        if (!snap.exists()) {
          showToast("Job no longer exists.");
          closeDislikeModal();
          return;
        }
        const jobData = snap.data();
        const prevDislikes = Number(jobData.dislikes || 0);
        const nextDislikes = prevDislikes + 1;
        const suggestions = Array.isArray(jobData.dislikeSuggestions) ? [...jobData.dislikeSuggestions] : [];
        suggestions.push({
          uid: currentUser.uid,
          loginId: currentUserData.loginId || currentUser.uid,
          suggestedAnswer: suggestion,
          createdAt: new Date().toISOString()
        });

        const authorUid = jobData.authorUID && jobData.authorUID !== "Admin" ? String(jobData.authorUID) : "";
        const displayJobId = jobData.jobId || docId;
        const displayJobName = jobData.jobName || "Job";

        if (nextDislikes >= 3) {
          await setDoc(doc(db, "archivedJobs", docId), {
            ...jobData,
            id: docId,
            dislikes: nextDislikes,
            dislikeSuggestions: suggestions,
            archived: true,
            archivedReason: "3_dislikes",
            archiveStatus: "archived",
            archivedAt: new Date().toISOString(),
            archivedBy: currentUser.uid
          });
          await deleteDoc(jobRef);

          if (authorUid) {
            await create48hNotification({
              title: `3rd Dislike: Job Archived (${displayJobId})`,
              message: `Your answer for "${displayJobName}" (Job ID: ${displayJobId}) received its 3rd dislike and has been moved to archive for review. Latest suggestion: "${suggestion}".`,
              targetUid: authorUid,
              eventType: "third_dislike_archived",
              meta: { jobId: displayJobId, dislikes: nextDislikes }
            });
          }

          await create48hNotification({
            title: `3rd Dislike Alert: Job ${displayJobId} Archived`,
            message: `Job "${displayJobName}" (ID: ${displayJobId}) reached 3 dislikes and was archived.`,
            targetUid: ADMIN_UID,
            eventType: "third_dislike_admin",
            meta: { jobId: displayJobId, dislikes: nextDislikes }
          });

          showToast("3rd dislike recorded — job moved to archive & notification sent.");
        } else {
          await updateDoc(jobRef, {
            dislikes: nextDislikes,
            dislikeSuggestions: suggestions,
            updatedAt: new Date().toISOString()
          });

          if (authorUid) {
            await create48hNotification({
              title: `Dislike #${nextDislikes} on Job ${displayJobId}`,
              message: `Your answer for "${displayJobName}" (Job ID: ${displayJobId}) received dislike #${nextDislikes} out of 3. Suggested answer: "${suggestion}".`,
              targetUid: authorUid,
              eventType: "each_dislike",
              meta: { jobId: displayJobId, dislikeNumber: nextDislikes }
            });
          }

          showToast(`Dislike #${nextDislikes} recorded & notification sent.`);
        }

        closeDislikeModal();
        await triggerSearch();
      } catch (err) {
        showToast("Could not submit dislike: " + (err.message || err));
      } finally {
        isSubmittingDislike = false;
        btn.disabled = false;
      }
    };

    window.submitNewAnswer = async function(e) {
      e.preventDefault();
      if (isSubmittingAnswer || !currentUser) return;

      const jobName = document.getElementById("post-job-name").value.trim();
      const jobIdRaw = document.getElementById("post-job-id").value.trim();
      const answer = document.getElementById("post-answer-text").value.trim();
      const imageUrl = document.getElementById("post-img-url").value.trim();
      const youtubeLink = document.getElementById("post-yt-url").value.trim();

      const expectedLen = Math.max(1, Number(formBuilderConfig.jobIdLength) || 5);
      if (jobIdRaw.length > expectedLen) {
        showToast(`Job ID must not exceed ${expectedLen} characters.`);
        return;
      }
      if (formBuilderConfig.jobIdDigitsOnly && !/^\d+$/.test(jobIdRaw)) {
        showToast("Job ID must contain numbers only.");
        return;
      }

      const customData = {};
      document.querySelectorAll(".admin-custom-field-input").forEach(inp => {
        const lbl = inp.dataset.fieldLabel || "Custom Field";
        const fType = inp.dataset.fieldType || "text";
        const val = inp.value.trim();
        if (fType === "number" && val && !/^\d+$/.test(val)) {
          throw new Error(`${lbl} must contain numbers only.`);
        }
        if (val) customData[lbl] = val;
      });

      isSubmittingAnswer = true;
      const btn = document.getElementById("postAnswerSubmitBtn");
      btn.disabled = true;

      try {
        const cleanDocId = jobIdRaw.replace(/[\/\\?#%]/g, "_");
        const loginId = currentUserData.loginId || currentUser.email?.split("@")[0] || currentUser.uid;
        const payload = {
          id: cleanDocId,
          jobId: jobIdRaw,
          jobName,
          answer,
          image: imageUrl,
          imageUrl,
          youtubeLink,
          customFields: customData,
          authorUID: currentUser.uid,
          authorLoginId: loginId,
          visible: true,
          archived: false,
          likes: 0,
          dislikes: 0,
          dislikeSuggestions: [],
          createdAt: new Date().toISOString(),
          updatedAt: new Date().toISOString()
        };

        await setDoc(doc(db, "jobs", cleanDocId), payload);

        const userDocRef = doc(db, "users", currentUserDocId || currentUser.uid);
        const nextSubmitted = Number(currentUserData.submittedCount || 0) + 1;
        await updateDoc(userDocRef, {
          submittedCount: increment(1),
          updatedAt: new Date().toISOString()
        });
        currentUserData.submittedCount = nextSubmitted;

        await loadAllJobs();
        renderProfileStatsUI(computeUserStats(currentUserData, allJobsCache));

        e.target.reset();
        document.getElementById("post-img-url").value = "";
        const prev = document.getElementById("post-img-preview");
        if (prev) { prev.src = ""; prev.style.display = "none"; }

        savePendingJobReloadState(jobIdRaw, jobName);
        showJobReloadPopup(jobIdRaw, jobName);
        showToast("Answer posted successfully!");
      } catch (err) {
        showToast("Could not post answer: " + (err.message || err));
      } finally {
        isSubmittingAnswer = false;
        btn.disabled = false;
      }
    };

          <div class="my-ans-actions">
            <button type="button" class="update-btn" onclick="quickEditAnswer('${escapeHtml(j.id)}')">Edit Answer</button>
          </div>
        </div>
      `).join("");
    };

    window.quickEditAnswer = async function(docId) {
      const item = allJobsCache.find(j => j.id === docId);
      if (!item) return;
      const updated = prompt("Update answer:", item.answer || "");
      if (updated === null || !updated.trim()) return;
      try {
        await updateDoc(doc(db, "jobs", docId), {
          answer: updated.trim(),
          edited: true,
          editCount: increment(1),
          updatedAt: new Date().toISOString()
        });
        showToast("Answer updated.");
        await renderMyAnswers();
      } catch (err) {
        showToast("Update failed: " + (err.message || err));
      }
    };

    window.saveProfile = async function(e) {
      e.preventDefault();
      if (!currentUser) return;
      const name = document.getElementById("prof-name").value.trim();
      const age = document.getElementById("prof-age").value.trim();
      const nationality = document.getElementById("prof-nation").value.trim();
      const telegramUsername = document.getElementById("prof-telegram").value.trim();

      try {
        const ref = doc(db, "users", currentUserDocId || currentUser.uid);
        await setDoc(ref, {
          name, age, nationality, telegramUsername,
          updatedAt: new Date().toISOString()
        }, { merge: true });
        currentUserData = { ...currentUserData, name, age, nationality, telegramUsername };
        showToast("Profile updated.");
      } catch (err) {
        showToast("Could not update profile: " + (err.message || err));
      }
    };

    window.selectPayMethod = function(el, method) {
      selectedPayMethod = method;
      document.querySelectorAll(".pay-pill").forEach(p => p.classList.remove("active"));
      if (el) el.classList.add("active");
      updatePaymentNumberDisplay();
    };

    window.submitRenew = async function(e) {
      e.preventDefault();
      if (isSubmittingRenew || !currentUser) return;
      const senderMobile = document.getElementById("renew-sender-mobile").value.trim();
      const trxId = document.getElementById("renew-trx-id").value.trim();
      const amount = document.getElementById("ui-renew-amount").value.trim() || "500 BDT";

      if (!senderMobile || !trxId) {
        showToast("Please enter Sender Mobile Number and Transaction ID.");
        return;
      }

      isSubmittingRenew = true;
      try {
        await addDoc(collection(db, "renewRequests"), {
          uid: currentUser.uid,
          loginId: currentUserData.loginId || currentUser.email?.split("@")[0] || currentUser.uid,
          paymentMethod: selectedPayMethod,
          senderMobile,
          trxId,
          amount,
          status: "pending",
          createdAt: new Date().toISOString()
        });
        e.target.reset();
        await loadMyRenewHistory();
        showToast("Renewal request submitted!");
      } catch (err) {
        showToast("Could not submit renewal request: " + (err.message || err));
      } finally {
        isSubmittingRenew = false;
      }
    };

    let cropperInstance = null;
    window.previewAndCropImage = function(event) {
      const file = event.target.files?.[0];
      if (!file) return;
      const reader = new FileReader();
      reader.onload = () => {
        const cropImg = document.getElementById("cropImage");
        cropImg.src = reader.result;
        document.getElementById("cropModal").classList.add("open");
        if (cropperInstance) cropperInstance.destroy();
        if (window.Cropper) {
          cropperInstance = new window.Cropper(cropImg, { aspectRatio: 16 / 9, viewMode: 1 });
        }
      };
      reader.readAsDataURL(file);
    };

    window.cancelCrop = function() {
      document.getElementById("cropModal").classList.remove("open");
      if (cropperInstance) { cropperInstance.destroy(); cropperInstance = null; }
      const fi = document.getElementById("post-img-file");
      if (fi) fi.value = "";
    };

    window.applyCroppedImage = function() {
      let dataUrl = "";
      if (cropperInstance) {
        const canvas = cropperInstance.getCroppedCanvas({ width: 800, height: 450 });
        dataUrl = canvas ? canvas.toDataURL("image/jpeg", 0.85) : "";
      } else {
        dataUrl = document.getElementById("cropImage").src || "";
      }
      if (dataUrl) {
        document.getElementById("post-img-url").value = dataUrl;
        const prev = document.getElementById("post-img-preview");
        prev.src = dataUrl;
        prev.style.display = "block";
      }
      document.getElementById("cropModal").classList.remove("open");
      if (cropperInstance) { cropperInstance.destroy(); cropperInstance = null; }
      showToast("16:9 image applied.");
    };
