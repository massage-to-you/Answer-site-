    function escapeHtml(str) {

        .replace(/&/g, "&amp;")
        .replace(/</g, "&lt;")

      if (Number.isNaN(d.getTime())) return String(val);
      return new Intl.DateTimeFormat("en-GB", { timeZone: "Asia/Dhaka", dateStyle: "medium", timeStyle: "short", hour12: true }).format(d) + " (BDT)";
    }

  <script type="module">
    import { auth, db } from "./firebase.js";
    import { onAuthStateChanged, signInWithEmailAndPassword, signOut } from "https://www.gstatic.com/firebasejs/13.4.0/firebase-auth.js";
    import { collection, doc, getDoc, setDoc, updateDoc, deleteDoc, onSnapshot, query, orderBy } from "https://www.gstatic.com/firebasejs/13.4.0/firebase-firestore.js";

    const adminLoginScreen = document.getElementById("adminLoginScreen");
    const adminPanelWrapper = document.getElementById("adminPanelWrapper");
    const adminLoginForm = document.getElementById("adminLoginForm");
    const adminLoginError = document.getElementById("adminLoginError");
    const adminLoginSubmitBtn = document.getElementById("adminLoginSubmitBtn");

    function showAdminLoginErr(msg) {
      if (!adminLoginError) return;
      adminLoginError.textContent = msg;
      adminLoginError.style.display = "block";
    }

    if (adminLoginForm) {
      adminLoginForm.addEventListener("submit", async (e) => {
        e.preventDefault();
        if (adminLoginError) adminLoginError.style.display = "none";
        const email = document.getElementById("adminEmailInput").value.trim();
        const pass = document.getElementById("adminPasswordInput").value;
        if (!email || !pass) {
          showAdminLoginErr("Please enter both admin email and password.");
          return;
        }
        adminLoginSubmitBtn.disabled = true;
        adminLoginSubmitBtn.innerHTML = '<i class="fa-solid fa-spinner fa-spin"></i> Signing in...';
        try {
          const cred = await signInWithEmailAndPassword(auth, email, pass);
          if (!cred.user || cred.user.uid !== ADMIN_UID) {
            await signOut(auth);
            showAdminLoginErr("Access Denied: This account is not authorized as Admin.");
          }
        } catch (err) {
          showAdminLoginErr(err?.message || "Invalid email or password.");
        } finally {
          adminLoginSubmitBtn.disabled = false;
          adminLoginSubmitBtn.innerHTML = '<i class="fa-solid fa-right-to-bracket"></i> Login to Admin Panel';
        }
      });
    }

    let adminListenersStarted = false;

    onAuthStateChanged(auth, async (user) => {
      if (!user || user.uid !== ADMIN_UID) {
        if (adminPanelWrapper) adminPanelWrapper.style.display = "none";
        if (adminLoginScreen) adminLoginScreen.style.display = "flex";
        return;
      }
      if (adminLoginScreen) adminLoginScreen.style.display = "none";
      if (adminPanelWrapper) adminPanelWrapper.style.display = "block";
      if (adminListenersStarted) return;
      adminListenersStarted = true;

      const homeRef = doc(db, "siteSettings", "home");
      const jobFormRef = doc(db, "siteSettings", "jobForm");

      let currentHome = {};
      let currentJobForm = {};

      const [homeSnap, jobFormSnap] = await Promise.all([getDoc(homeRef), getDoc(jobFormRef)]);
      if (homeSnap.exists()) currentHome = homeSnap.data() || {};
      if (jobFormSnap.exists()) currentJobForm = jobFormSnap.data() || {};
      applyLoadedSettingsToInputs(currentHome, currentJobForm);

      onSnapshot(homeRef, (snap) => {
        currentHome = snap.exists() ? (snap.data() || {}) : {};
        applyLoadedSettingsToInputs(currentHome, currentJobForm);
      });

      // 1. Login history with 48-hour automatic deletion
      onSnapshot(collection(db, "loginHistory"), (snap) => {
        const nowMs = Date.now();
        const activeRows = [];
        for (const d of snap.docs) {
          const data = d.data() || {};
          const rawTime = data.loginAt || data.createdAt || data.timestamp || "";
          const ts = rawTime?.toMillis ? rawTime.toMillis() : (Date.parse(rawTime) || nowMs);
          if (nowMs - ts > FORTY_EIGHT_HOURS_MS) {
            deleteDoc(doc(db, "loginHistory", d.id)).catch(() => {});
            continue;
          }
          activeRows.push({ id: d.id, ...data, _ts: ts });
        }
        activeRows.sort((a, b) => b._ts - a._ts);
        renderLoginHistoryRows(activeRows);
      });

      // 2. Deposit / Withdraw requests
      onSnapshot(collection(db, "walletTransactions"), (snap) => {
        const rows = snap.docs.map(d => ({ id: d.id, ...d.data() }))
          .sort((a, b) => String(b.createdAt || "").localeCompare(String(a.createdAt || "")));
        renderWalletRequestsAdmin(rows);
      });

      window.reviewWalletRequest = async (reqId, nextStatus) => {
        if (!await adminConfirm(`Mark this request as ${nextStatus.toUpperCase()}?`, "Review Wallet Request")) return;
        await updateDoc(doc(db, "walletTransactions", reqId), {
          status: nextStatus,
          reviewedAt: new Date().toISOString(),
          reviewedBy: ADMIN_UID
        });
        showAdminToast(`Request marked ${nextStatus}.`);
      };

      // 3 & 6. Menu visibility + Logout selected / all users
      onSnapshot(collection(db, "users"), (snap) => {
        const users = snap.docs.map(d => ({ uid: d.id, ...(d.data() || {}) }));
        renderUsersDropdown(users);
      });

      window.saveMenuVisibilitySettings = async () => {
        const walletVisible = document.getElementById("menuVisWalletWd").checked;
        const menuVisibility = {
          profile: document.getElementById("menuVisProfile").checked,
          "post-answer": document.getElementById("menuVisPostAnswer").checked,
          "my-answer": document.getElementById("menuVisMyAnswer").checked,
          "top-users": document.getElementById("menuVisTopUsers").checked,
          notifications: document.getElementById("menuVisNotifications").checked,
          "wallet-wd": walletVisible,
          marketplace: document.getElementById("menuVisMarketplace").checked,
          renew: document.getElementById("menuVisRenew").checked,
          support: document.getElementById("menuVisSupport").checked,
          "support-ticket": document.getElementById("menuVisSupportTicket").checked
        };
        await setDoc(homeRef, {
          menuVisibility,
          showWalletSection: walletVisible,
          updatedAt: new Date().toISOString()
        }, { merge: true });
        showAdminToast("3-line menu visibility saved.");
      };

      window.logoutSelectedUser = async () => {
        const targetUid = document.getElementById("forceLogoutUserSelect").value;
        if (!targetUid) {
          showAdminToast("Please select a user first.");
          return;
        }
        if (!await adminConfirm("Force logout the selected user immediately?", "Logout Selected User")) return;
        await updateDoc(doc(db, "users", targetUid), {
          forceLogout: true,
          forceLogoutAt: Date.now()
        });
        showAdminToast("Selected user logged out.");
      };

      window.logoutAllUsers = async () => {
        if (!await adminConfirm("Force logout ALL users immediately?", "Logout All Users")) return;
        await setDoc(homeRef, {
          forceLogoutAllAt: Date.now()
        }, { merge: true });
        showAdminToast("Logout signal sent to all users.");
      };

      // 4. Save Form Builder
      window.saveFullFormBuilder = async () => {
        const jobIdLength = Math.max(1, Math.min(20, Number(document.getElementById("fbJobIdLength").value || 5)));
        const jobIdDigitsOnly = document.getElementById("fbJobIdMode").value !== "alnum";
        await setDoc(jobFormRef, {
          jobIdLength,
          jobIdDigitsOnly,
          standardFields: adminStandardFields,
          fields: adminCustomFields,
          updatedAt: new Date().toISOString()
        }, { merge: true });
        showAdminToast("Form Builder saved.");
      };

      // 5. Support tickets
      onSnapshot(query(collection(db, "supportTickets"), orderBy("createdAt", "desc")), (snap) => {
        const rows = snap.docs.map(d => ({ id: d.id, ...d.data() }));
        renderSupportTicketsAdmin(rows);
      }, () => {
        onSnapshot(collection(db, "supportTickets"), (snap) => {
          const rows = snap.docs.map(d => ({ id: d.id, ...d.data() }));
          renderSupportTicketsAdmin(rows);
        });
      });

      // 7. Save mobile banking numbers & scrolling headline
      window.saveWalletBankingAndHeadline = async () => {
        const walletHeadline = document.getElementById("wdScrollingHeadlineInput").value.trim();
        const bkashNumber = document.getElementById("wdBkashNumberInput").value.trim();
        const nagadNumber = document.getElementById("wdNagadNumberInput").value.trim();
        const rocketNumber = document.getElementById("wdRocketNumberInput").value.trim();
        await setDoc(homeRef, {
          walletHeadline,
          bkashNumber,
          nagadNumber,
          rocketNumber,
          updatedAt: new Date().toISOString()
        }, { merge: true });
        showAdminToast("Mobile banking numbers & headline saved.");
      };

      window.logoutAdminAccount = async () => {
        await signOut(auth);
        if (adminPanelWrapper) adminPanelWrapper.style.display = "none";
        if (adminLoginScreen) adminLoginScreen.style.display = "flex";
      };
    });
