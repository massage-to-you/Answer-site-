async function toggleUser(id){const u=firestoreUsers.find(x=>x.id===id);if(!u)return;const blocked=["blocked","inactive"].includes(String(u.status||"active").toLowerCase());if(!confirm(`Are you sure you want to ${blocked?"activate":"deactivate"} this user?`))return;try{await updateDoc(doc(db,"users",id),{status:blocked?"active":"blocked",statusUpdatedAt:new Date().toISOString()});await loadUsers();if(selectedUserId===id)openProfile(id);}catch(err){alert("Status update failed: "+(err.code||err.message));}}

async function loadJobs(){
 $("jobsEmpty").textContent="Loading jobs...";$("jobsEmpty").classList.remove("hidden");
 try{const snap=await getDocs(collection(db,"jobs"));firestoreJobs=snap.docs.map(d=>({id:d.id,...d.data()}));$("totalJobs").textContent=firestoreJobs.length;const recent=[...firestoreJobs].slice(-5).reverse();$("recentJobsBody").innerHTML=recent.map(j=>`<tr><td>${esc(j.id)}</td><td>${esc(j.jobName||"—")}</td><td>${esc(j.authorUID||"—")}</td></tr>`).join("")||'<tr><td colspan="3" class="empty">No jobs found.</td></tr>';renderJobs();}
 catch(err){firestoreJobs=[];$("totalJobs").textContent="!";$("jobsBody").innerHTML="";$("jobsEmpty").textContent="Could not load jobs: "+(err.code||err.message);$("jobsEmpty").classList.remove("hidden");}
}
function renderJobs(){const q=$("jobSearch").value.toLowerCase().trim(),list=firestoreJobs.filter(j=>`${j.id} ${j.jobName||""} ${j.answer||""}`.toLowerCase().includes(q));$("jobsBody").innerHTML=list.map(j=>`<tr><td>${esc(j.id)}</td><td>${esc(j.jobName||"—")}</td><td>${esc(j.answer||"—")}</td><td>${esc(j.authorUID||"—")}</td><td><button class="btn secondary" data-edit-job="${esc(j.id)}">Edit</button> <button class="btn danger" data-delete-job="${esc(j.id)}">Delete</button></td></tr>`).join("");$("jobsEmpty").classList.toggle("hidden",list.length>0);if(!list.length)$("jobsEmpty").textContent="No jobs found.";}
async function addJob(){const raw=prompt("Enter unique Job ID:");if(raw===null)return;const id=raw.trim();if(!id)return alert("Job ID is required.");if(firestoreJobs.some(j=>j.id===id))return alert("Job ID already exists.");const name=prompt("Enter Job Name:");if(name===null)return;const answer=prompt("Enter Answer:");if(answer===null)return;try{await setDoc(doc(db,"jobs",id),{jobName:name.trim(),answer:answer.trim(),dislikes:0,visible:true,authorUID:"Admin",edited:false,editCount:0,createdAt:new Date().toISOString()});await loadJobs();alert("Job added.");}catch(err){alert("Add failed: "+(err.code||err.message));}}
async function editJob(id){const j=firestoreJobs.find(x=>x.id===id);if(!j)return;const name=prompt("Edit Job Name:",j.jobName||"");if(name===null)return;const answer=prompt("Edit Answer:",j.answer||"");if(answer===null)return;try{await updateDoc(doc(db,"jobs",id),{jobName:name.trim(),answer:answer.trim(),edited:true,editCount:increment(1),updatedAt:new Date().toISOString()});await loadJobs();}catch(err){alert("Edit failed: "+(err.code||err.message));}}
async function deleteJob(id){if(!confirm("Permanently delete Job ID "+id+"?"))return;try{await deleteDoc(doc(db,"jobs",id));await loadJobs();}catch(err){alert("Delete failed: "+(err.code||err.message));}}
$("jobsBody").addEventListener("click",e=>{const a=e.target.closest("[data-edit-job]"),d=e.target.closest("[data-delete-job]");if(a)editJob(a.dataset.editJob);if(d)deleteJob(d.dataset.deleteJob);});
