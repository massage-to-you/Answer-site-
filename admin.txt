  <script type="module">
    import { initializeApp } from "https://www.gstatic.com/firebasejs/10.12.2/firebase-app.js";
    import { getAuth, onAuthStateChanged } from "https://www.gstatic.com/firebasejs/10.12.2/firebase-auth.js";
    import {
      getFirestore,
      doc,
      getDoc,
      setDoc,
      addDoc,
      updateDoc,
      deleteDoc,
      collection,
      getDocs,
      orderBy,
      query,
      serverTimestamp,
      Timestamp
    } from "https://www.gstatic.com/firebasejs/10.12.2/firebase-firestore.js";

    // =========================================================
    // REPLACE WITH YOUR FIREBASE PROJECT CONFIG
    // =========================================================
    const firebaseConfig = {
      apiKey: "YOUR_API_KEY",
      authDomain: "answer-site.firebaseapp.com",
      projectId: "answer-site",
      storageBucket: "answer-site.appspot.com",
      messagingSenderId: "YOUR_MESSAGING_SENDER_ID",
      appId: "YOUR_APP_ID"
    };

    const ADMIN_UID = "DEBuc9USNcULaH9BY1tXvbRJZYn1";
    const FORTY_EIGHT_HOURS_MS = 48 * 60 * 60 * 1000;

    const app = initializeApp(firebaseConfig);
    const auth = getAuth(app);
    const db = getFirestore(app);

    const loader = document.getElementById("adminLoader");
    function showLoader() { loader.classList.remove("hidden"); }
    function hideLoader() { loader.classList.add("hidden"); }

    function showToast(msg) {
      const box = document.getElementById("toastBox");
      box.textContent = msg;
      box.style.display = "block";
      clearTimeout(window._t);
      window._t = setTimeout(() => { box.style.display = "none"; }, 3000);
    }

    window.openTab = function(tabId, btn) {
      document.querySelectorAll(".tab-panel").forEach(p => p.classList.remove("active"));
      document.getElementById(tabId).classList.add("active");
      document.querySelectorAll(".tab-btn").forEach(b => b.classList.remove("active"));
      btn.classList.add("active");
    };

    function fmtMoney(val) {
      const n = Number(val);
      return (isNaN(n) ? 0 : n).toFixed(2);
    }

    function escapeHtml(str) {
      return String(str)
        .replace(/&/g, "&amp;")
        .replace(/</g, "&lt;")
        .replace(/>/g, "&gt;")
        .replace(/"/g, "&quot;");
    }

    async function loadSettings() {
      const snap = await getDoc(doc(db, "siteSettings", "home"));
      const d = snap.exists() ? snap.data() : {};
      document.getElementById("rewardRateInput").value = fmtMoney(d.rewardPerSuccess ?? 0.50);
      document.getElementById("deductRateInput").value = fmtMoney(d.deductionPerFail ?? 0.25);
      document.getElementById("renewCostInput").value = fmtMoney(d.depositRenewCost ?? 4.00);
      document.getElementById("noticeInput").value = d.notice || "Welcome to Answer Site!";
      document.getElementById("guideInput").value = d.userGuide || "";
      document.getElementById("rulesInput").value = d.websiteRules || "";
    }

    document.getElementById("settingsForm").addEventListener("submit", async (e) => {
      e.preventDefault();
      showLoader();
      try {
        await setDoc(doc(db, "siteSettings", "home"), {
          rewardPerSuccess: Number(document.getElementById("rewardRateInput").value),
          deductionPerFail: Number(document.getElementById("deductRateInput").value),
          depositRenewCost: Number(document.getElementById("renewCostInput").value),
          notice: document.getElementById("noticeInput").value.trim(),
          userGuide: document.getElementById("guideInput").value.trim(),
          websiteRules: document.getElementById("rulesInput").value.trim(),
          updatedAt: serverTimestamp()
        }, { merge: true });
        showToast("Settings updated successfully!");
      } catch (err) {
        showToast("Error: " + err.message);
      } finally {
        hideLoader();
      }
    });

    async function loadUsers() {
      const container = document.getElementById("usersListContainer");
      const snap = await getDocs(collection(db, "users"));
      if (snap.empty) {
        container.innerHTML = `<p style="font-size:13px; color:#94a3b8;">No users found.</p>`;
        return;
      }
      container.innerHTML = snap.docs.map(d => {
        const u = d.data();
        return `
          <div class="list-item">
            <div style="display:flex; justify-content:space-between; flex-wrap:wrap; gap:6px; margin-bottom:8px;">
              <div>
                <strong style="color:#ffffff;">${escapeHtml(u.name || "Unnamed")}</strong>
                <span style="font-size:11px; color:#38bdf8; margin-left:6px;">UID: ${d.id}</span>
              </div>
              <span style="font-size:12px; color:#fbbf24;">Deposit: ${fmtMoney(u.depositBalance || 0)}</span>
            </div>
            <div class="grid-2" style="margin-bottom:8px;">
              <div>
                <label style="font-size:11px; color:#94a3b8;">Deposit Balance</label>
                <input type="number" step="0.01" id="dep_${d.id}" class="form-control" value="${fmtMoney(u.depositBalance || 0)}" />
              </div>
              <div>
                <label style="font-size:11px; color:#94a3b8;">Valid Until (YYYY-MM-DD)</label>
                <input type="text" id="val_${d.id}" class="form-control" value="${escapeHtml(u.validUntil || "")}" />
              </div>
              <div>
                <label style="font-size:11px; color:#94a3b8;">Successful Answers</label>
                <input type="number" id="suc_${d.id}" class="form-control" value="${Number(u.successfulAnswers || 0)}" />
              </div>
              <div>
                <label style="font-size:11px; color:#94a3b8;">Unsuccessful Answers</label>
                <input type="number" id="uns_${d.id}" class="form-control" value="${Number(u.unsuccessfulAnswers || 0)}" />
              </div>
            </div>
            <button class="btn btn-primary" onclick="saveUserChanges('${d.id}')">
              <i class="fa-solid fa-check"></i> Update User
            </button>
          </div>
        `;
      }).join("");
    }

    window.saveUserChanges = async function(uid) {
      showLoader();
      try {
        await updateDoc(doc(db, "users", uid), {
          depositBalance: Number(document.getElementById(`dep_${uid}`).value),
          validUntil: document.getElementById(`val_${uid}`).value.trim(),
          successfulAnswers: Number(document.getElementById(`suc_${uid}`).value),
          unsuccessfulAnswers: Number(document.getElementById(`uns_${uid}`).value),
          updatedAt: serverTimestamp()
        });
        showToast("User updated!");
        await loadUsers();
      } catch (err) {
        showToast("Error: " + err.message);
      } finally {
        hideLoader();
      }
    };

    // Notifications (48-hour expiry)
    document.getElementById("sendNotifForm").addEventListener("submit", async (e) => {
      e.preventDefault();
      showLoader();
      try {
        const nowMs = Date.now();
        const expiresAt = Timestamp.fromMillis(nowMs + FORTY_EIGHT_HOURS_MS);
        await addDoc(collection(db, "notifications"), {
          title: document.getElementById("notifTitleInput").value.trim(),
          targetUid: document.getElementById("notifTargetInput").value.trim() || "all",
          message: document.getElementById("notifMsgInput").value.trim(),
          createdAt: Timestamp.fromMillis(nowMs),
          expiresAt
        });
        e.target.reset();
        document.getElementById("notifTargetInput").value = "all";
        showToast("Notification published (auto-deletes in 48h)!");
        await loadAdminNotifications();
      } catch (err) {
        showToast("Error: " + err.message);
      } finally {
        hideLoader();
      }
    });

    async function loadAdminNotifications() {
      const container = document.getElementById("adminNotifList");
      const q = query(collection(db, "notifications"), orderBy("createdAt", "desc"));
      const snap = await getDocs(q);
      const nowMs = Date.now();
      const active = [];

      for (const d of snap.docs) {
        const data = d.data();
        const expiresMs = data.expiresAt?.toMillis ? data.expiresAt.toMillis() : (nowMs + FORTY_EIGHT_HOURS_MS);
        if (expiresMs <= nowMs) {
          await deleteDoc(doc(db, "notifications", d.id)).catch(() => {});
          continue;
        }
        active.push({ id: d.id, ...data, expiresMs });
      }

      if (active.length === 0) {
        container.innerHTML = `<p style="font-size:13px; color:#94a3b8;">No active notifications.</p>`;
        return;
      }

      container.innerHTML = active.map(n => {
        const hrsLeft = Math.max(1, Math.ceil((n.expiresMs - nowMs) / (1000 * 60 * 60)));
        return `
          <div class="list-item" style="display:flex; justify-content:space-between; align-items:center; gap:10px;">
            <div>
              <strong style="color:#38bdf8;">${escapeHtml(n.title)}</strong>
              <span style="font-size:11px; color:#fbbf24; margin-left:6px;">Target: ${escapeHtml(n.targetUid || "all")} (${hrsLeft}h left)</span>
              <p style="font-size:13px; color:#e2e8f0; margin-top:4px;">${escapeHtml(n.message)}</p>
            </div>
            <button class="btn btn-danger" onclick="deleteNotif('${n.id}')">
              <i class="fa-solid fa-trash"></i>
            </button>
          </div>
        `;
      }).join("");
    }

    window.deleteNotif = async function(id) {
      showLoader();
      await deleteDoc(doc(db, "notifications", id));
      await loadAdminNotifications();
      hideLoader();
      showToast("Notification deleted.");
    };

    // Renew Requests
    async function loadRenewRequests() {
      const container = document.getElementById("adminRenewList");
      const snap = await getDocs(collection(db, "renewRequests"));
      const items = snap.docs.map(d => ({ id: d.id, ...d.data() }));
      items.sort((a, b) => (b.createdAt?.toMillis?.() || 0) - (a.createdAt?.toMillis?.() || 0));

      if (items.length === 0) {
        container.innerHTML = `<p style="font-size:13px; color:#94a3b8;">No renewal requests.</p>`;
        return;
      }

      container.innerHTML = items.map(r => {
        const status = (r.status || "pending").toLowerCase();
        const dateStr = r.createdAt?.toDate ? r.createdAt.toDate().toLocaleString() : "";
        return `
          <div class="list-item">
            <div style="display:flex; justify-content:space-between; flex-wrap:wrap; gap:6px;">
              <strong>${escapeHtml(r.userName || r.uid)} — ${escapeHtml(r.method || "Manual")} (${fmtMoney(r.amount || 0)})</strong>
              <span style="font-size:12px; color:#38bdf8; font-weight:700;">${status.toUpperCase()}</span>
            </div>
            <div style="font-size:12px; color:#94a3b8; margin:4px 0 8px;">
              TrxID: ${escapeHtml(r.trxId || "-")} • UID: ${escapeHtml(r.uid || "")} • ${dateStr}
            </div>
            <div style="display:flex; gap:8px;">
              ${status === "pending" ? `
                <button class="btn btn-success" onclick="updateRenewStatus('${r.id}', 'approved')">Approve</button>
                <button class="btn btn-danger" onclick="updateRenewStatus('${r.id}', 'rejected')">Reject</button>
              ` : ""}
              <button class="btn btn-outline" onclick="deleteRenewReq('${r.id}')">Delete</button>
            </div>
          </div>
        `;
      }).join("");
    }

    window.updateRenewStatus = async function(id, status) {
      showLoader();
      await updateDoc(doc(db, "renewRequests", id), { status });
      await loadRenewRequests();
      hideLoader();
      showToast(`Request marked as ${status}.`);
    };

    window.deleteRenewReq = async function(id) {
      showLoader();
      await deleteDoc(doc(db, "renewRequests", id));
      await loadRenewRequests();
      hideLoader();
      showToast("Record deleted.");
    };

    onAuthStateChanged(auth, async (user) => {
      if (user && user.uid === ADMIN_UID) {
        document.getElementById("deniedBox").style.display = "none";
        document.getElementById("adminApp").style.display = "block";
        await Promise.all([
          loadSettings(),
          loadUsers(),
          loadAdminNotifications(),
          loadRenewRequests()
        ]);
        hideLoader();
      } else {
        document.getElementById("adminApp").style.display = "none";
        document.getElementById("deniedBox").style.display = "flex";
        hideLoader();
      }
    });
