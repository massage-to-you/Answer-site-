<script type="module">
import { auth } from "./firebase.js";
import {
  signInWithEmailAndPassword,
  onAuthStateChanged,
  signOut
} from "https://www.gstatic.com/firebasejs/13.4.0/firebase-auth.js";

const ADMIN_UID = "DEBuc9USNcULaH9BY1tXvbRJZYn1";
const $ = id => document.getElementById(id);
const read = (key, fallback) => {
  try { return JSON.parse(localStorage.getItem(key)) ?? fallback; }
  catch { return fallback; }
};

let authCheckFinished = false;

function showLogin(message = "") {
  $("dashboardView").classList.add("hidden");
  $("loginView").classList.remove("hidden");
  if (message) {
    $("loginStatus").className = "status error";
    $("loginStatus").textContent = message;
  }
}

function showDashboard() {
  $("loginView").classList.add("hidden");
  $("dashboardView").classList.remove("hidden");
  refreshAll();
}

$("adminLoginForm").addEventListener("submit", async event => {
  event.preventDefault();

  const email = $("adminEmail").value.trim().toLowerCase();
  const password = $("adminPassword").value;
  const button = $("adminLoginButton");

  $("loginStatus").className = "status";
  $("loginStatus").textContent = "Checking administrator account...";
  button.disabled = true;
  button.textContent = "PLEASE WAIT...";

  try {
    const credential = await signInWithEmailAndPassword(auth, email, password);

    if (credential.user.uid !== ADMIN_UID) {
      await signOut(auth);
      showLogin("Access denied. This account is not the authorized administrator.");
      return;
    }

    $("loginStatus").className = "status success";
    $("loginStatus").textContent = "Administrator verified.";
    showDashboard();
  } catch (error) {
    let message = "Login failed. Check the email and password.";
    if (error.code === "auth/invalid-credential" ||
        error.code === "auth/wrong-password" ||
        error.code === "auth/user-not-found") {
      message = "Incorrect admin email or password.";
    } else if (error.code === "auth/too-many-requests") {
      message = "Too many attempts. Please wait and try again later.";
    } else if (error.code === "auth/network-request-failed") {
      message = "Network error. Check your internet connection.";
    }
    showLogin(message);
  } finally {
    button.disabled = false;
    button.textContent = "LOGIN TO ADMIN";
  }
});

$("logoutBtn").addEventListener("click", async () => {
  try {
    await signOut(auth);
    showLogin("You have been logged out.");
  } catch (error) {
    alert("Logout failed. Please try again.");
  }
});

/* A signed-in non-admin user must never see the admin dashboard. */
onAuthStateChanged(auth, async user => {
  authCheckFinished = true;

  if (!user) {
    showLogin();
    return;
  }

  if (user.uid !== ADMIN_UID) {
    await signOut(auth);
    showLogin("Access denied. Please sign in with the administrator account.");
    return;
  }

  showDashboard();
});

function go(section) {
  document.querySelectorAll(".nav-btn").forEach(button =>
    button.classList.toggle("active", button.dataset.section === section)
  );
  document.querySelectorAll(".section").forEach(item =>
    item.classList.toggle("active", item.id === "section-" + section)
  );
  window.scrollTo({ top: 0, behavior: "smooth" });
}

document.querySelectorAll(".nav-btn").forEach(button =>
  button.addEventListener("click", () => go(button.dataset.section))
);
document.querySelectorAll("[data-go]").forEach(button =>
  button.addEventListener("click", () => go(button.dataset.go))
);

function refreshAll() {
  const jobs = read("adminAnswersDB", []);
  const challenges = read("answerSiteChallenges", []);
  const deleted = read("answerSiteDeletionLog", []);
  const ledger = read("answerSiteSubmissionLedger", []);

  $("totalUsers").textContent = "—";
  $("activeUsers").textContent = "—";
  $("totalJobs").textContent = jobs.length;
  $("pendingChallenges").textContent =
    challenges.filter(item => (item.status || "PENDING") === "PENDING").length;
  $("deletedJobs").textContent = deleted.length;

  const today = new Date().toDateString();
  $("todayActivity").textContent =
    ledger.filter(item => item.createdAt && new Date(item.createdAt).toDateString() === today).length +
    deleted.filter(item => item.deletedAt && new Date(item.deletedAt).toDateString() === today).length;

  $("recentJobsBody").innerHTML = jobs.slice(-5).reverse().map(job => `
    <tr>
      <td>${escapeHTML(job.jobId || job.id || "—")}</td>
      <td>${escapeHTML(job.jobName || job.name || "—")}</td>
      <td>${escapeHTML(job.uid || job.submitterUID || "—")}</td>
      <td>${escapeHTML(job.createdAt || job.timestamp || "—")}</td>
    </tr>
  `).join("") || '<tr><td colspan="4" class="empty">No local job records.</td></tr>';

  renderJobs();

  const config = read("adminSiteConfig", {});
  const payments = config.paymentNumbers || {};
  $("payBkash").value = payments.bkash || "";
  $("payNagad").value = payments.nagad || "";
  $("payRocket").value = payments.rocket || "";
  $("headlineText").value = config.headline || "";
  $("adminWhatsapp").value = config.adminWhatsapp || "";
  $("adminTelegram").value = config.adminTelegram || "";
}

function escapeHTML(value) {
  return String(value).replace(/[&<>"']/g, character => ({
    "&": "&amp;",
    "<": "&lt;",
    ">": "&gt;",
    '"': "&quot;",
    "'": "&#39;"
  })[character]);
}

function renderJobs() {
  const query = ($("jobSearch").value || "").toLowerCase();
  const jobs = read("adminAnswersDB", []).filter(job =>
    JSON.stringify(job).toLowerCase().includes(query)
  );

  $("jobsBody").innerHTML = jobs.map(job => `
    <tr>
      <td>${escapeHTML(job.jobId || job.id || "—")}</td>
      <td>${escapeHTML(job.jobName || job.name || "—")}</td>
      <td>${escapeHTML(job.answer || "—")}</td>
      <td>${escapeHTML(job.uid || job.submitterUID || "—")}</td>
      <td>${escapeHTML(job.createdAt || job.timestamp || "—")}</td>
    </tr>
  `).join("");

  $("jobsEmpty").classList.toggle("hidden", jobs.length > 0);
}

$("jobSearch").addEventListener("input", renderJobs);
$("refreshJobs").addEventListener("click", refreshAll);

  alert("User creation is not enabled yet. It requires a trusted backend using Firebase Admin SDK.")

["savePayments", "saveWebsite"].forEach(id => {
  $(id).addEventListener("click", () => {
    const target = id === "savePayments" ? "paymentStatus" : "websiteStatus";
