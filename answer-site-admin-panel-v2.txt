function renderLogs(){
 const filter=$("logFilter").value, dels=getDeletions(), ledger=getLedger();
 let rows=[...dels.map(x=>({event:"DELETED",jobId:x.jobId,uid:x.uid||x.adminUID,reason:x.reason,date:x.deletedAt})),...ledger.map(x=>({event:"SUBMITTED",jobId:x.jobId,uid:x.uid,reason:"NEW_SUBMISSION",date:x.createdAt}))];
 if(filter==="deleted")rows=rows.filter(x=>x.event==="DELETED");if(filter==="submitted")rows=rows.filter(x=>x.event==="SUBMITTED");
 rows.sort((a,b)=>new Date(b.date||0)-new Date(a.date||0));
 $("logsBody").innerHTML=rows.map(x=>`<tr><td><span class="tag ${x.event==="DELETED"?"red":"green"}">${x.event}</span></td><td>${esc(x.jobId||"—")}</td><td>${esc(x.uid||"—")}</td><td>${esc(x.reason||"—")}</td><td>${esc(dateText(x.date))}</td></tr>`).join("");
 $("logsEmpty").classList.toggle("hidden",rows.length>0);
}
